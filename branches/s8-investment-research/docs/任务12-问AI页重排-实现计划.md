async function checkStock(){
  const code = ($("check-code").value || "").trim();
  if(!/^\d{6}$/.test(code)){ toast("请输入6位股票代码,如 600519"); return; }
  // 拍板:池内票 → 自动切回「我的课堂」并选中,不出重复内容
  if(POOL.find(r=>r.code===code)){
    showSub("learn"); selectStock(code);
    toast("这只票在您的池子里,已带您回课堂"); return;
  }
  CHECK_CODE = code;
  const seq = ++CHECK_SEQ;
  $("checkup-wrap").innerHTML = `<div class="checkup"><div class="note">体检中(拉K线+跑规则,约2~5秒)…</div></div>`;
  $("check-chart-wrap").innerHTML = `<div class="chartbox"><div id="kchart2"><span class="empty">K线加载中…</span></div>
    <div class="legend">池外票没有平台价/止损/入场点标注,这里只是原始K线</div></div>`;
  checkChatReset();
  drawChart(code, "kchart2");           // K线与体检并行拉
  let d;
  try{
    d = await (await fetch(`/api/stock_test/${code}`, {cache:"no-store"})).json();
  }catch(e){
    if(seq !== CHECK_SEQ) return;
    $("checkup-wrap").innerHTML = `<div class="checkup"><div class="note">网络异常,请稍后重试</div></div>`;
    return;
  }
  if(seq !== CHECK_SEQ) return;
  if(!d.ok){
    $("checkup-wrap").innerHTML = `<div class="checkup"><div class="note">${esc(d.error||"体检失败")}</div></div>`;
    return;
  }
  renderCheckup(d);                      // 沿用现 L566-609,删「就这只票问AI」按钮一行
  checkAsk(code, seq);                   // AI 白话解读自动发,不用点
}

function checkChatReset(){
  $("check-chat-wrap").innerHTML = `<div class="chat">
    <div id="check-msgs"></div>
    <div class="inputbar">
      <input id="check-input" placeholder="就这只票追问…">
      <button id="check-send" onclick="checkSend()">发送</button>
    </div>
  </div>`;
}

async function checkAsk(code, seq){
  const msg = `【池外查询 ${code}】请用大白话分析这只股票的行业与近期走势,`
    + `只基于你已有的注入数据,没有的信息明说没有。`;
  const thinking = appendMsg("check-msgs", "ai", `<div class="who">AI 助教 · 白话解读</div>AI 思考中…`);
  try{
    const d = await (await fetch("/api/review/chat", {
      method:"POST", headers:{"Content-Type":"application/json"},
      body: JSON.stringify({message: msg}),
    })).json();
    if(!d.ok) throw new Error(d.error || "引擎不可用");
    if(seq !== CHECK_SEQ) return;
    thinking.innerHTML = `<div class="who">AI 助教 · 池外票 ${esc(code)}</div>${esc(d.reply)}`;
  }catch(e){
    if(seq !== CHECK_SEQ) return;
    thinking.classList.add("err");
    thinking.innerHTML = `<div class="who">AI 助教</div>回答失败:${esc(e.message||String(e))}`;
  }
}

async function checkSend(){
  const input = $("check-input");
  const msg = (input.value||"").trim();
  if(!msg || !CHECK_CODE) return;
  $("check-send").disabled = true;
  input.value = "";
  appendMsg("check-msgs", "user", esc(msg));
  const thinking = appendMsg("check-msgs", "ai", `<div class="who">AI 助教</div>AI 思考中…`);
  try{
    const d = await (await fetch("/api/review/chat", {
      method:"POST", headers:{"Content-Type":"application/json"},
      body: JSON.stringify({message: `【池外查询 ${CHECK_CODE}】${msg}`}),
    })).json();
    if(!d.ok) throw new Error(d.error || "引擎不可用");
    thinking.innerHTML = `<div class="who">AI 助教 · 池外票 ${esc(CHECK_CODE)}</div>${esc(d.reply)}`;
  }catch(e){
    thinking.classList.add("err");
    thinking.innerHTML = `<div class="who">AI 助教</div>回答失败:${esc(e.message||String(e))}`;
  }
  $("check-send").disabled = false;
}
