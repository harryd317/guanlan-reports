```python
#!/usr/bin/env python3
"""P0 策略中心存量迁移:pool/positions/trades 补 strategy 归属 + pool 补 industry。

用法:
  .venv/bin/python migrate_strategy.py            # dry-run 只打印
  .venv/bin/python migrate_strategy.py --apply    # sqlite3 backup 备份后真实写库

归属规则(设计-策略中心-20260711 §3.4):按行的锚定日期(pool=entry_date 或 first_seen,
positions/trades=buy_date)取当日快照 payload.mode;mode∈{mid,short}→照写,
无快照/无 mode→'unknown'(unknown 永不进背书矩阵)。幂等:已有值的行跳过。
industry 回填:data.get_industry(code),拿不到留空并进对账报告。
"""
import argparse
import datetime as dt
import os
import sqlite3

import data
import storage

BASE = os.path.dirname(os.path.abspath(__file__))


def _snap_mode(date_str):
    if not date_str:
        return None
    snap = storage.load_snapshot(str(date_str)[:10])
    return (snap or {}).get("mode")


def _strategy_of(date_str):
    m = _snap_mode(date_str)
    return m if m in ("mid", "short") else "unknown"


def main():
    ap = argparse.ArgumentParser(description="策略归属/行业存量迁移(P0)")
    ap.add_argument("--apply", action="store_true", help="真实写库(默认 dry-run 只打印)")
    a = ap.parse_args()
    storage.init_db()   # 幂等建列:新列必须先于任何写入存在(不依赖服务先重启,任务14教训)
    if a.apply:
        db = os.path.join(BASE, "screener.db")
        bak = db + ".bak-" + dt.datetime.now().strftime("%Y%m%d_%H%M")
        # sqlite3 backup API:自带一致性快照,WAL 已提交未 checkpoint 的事务一并拷入
        src, dst = sqlite3.connect(db), sqlite3.connect(bak)
        with dst:
            src.backup(dst)
        src.close()
        dst.close()
        print(f"[备份] {bak}")

    n_pool = n_pos = n_trd = n_unknown = 0
    industry_miss = []

    for row in storage.pool_list():
        rid, code = row["id"], row["code"]
        name, state = row.get("name") or "", row.get("state")
        if row.get("strategy") and row.get("industry"):
            print(f"  ○ {code} {name}[{state}] 已有归属({row.get('strategy')}),跳过")
            continue
        strat = row.get("strategy") or _strategy_of(row.get("entry_date") or row.get("first_seen"))
        ind = row.get("industry") or data.get_industry(code)
        if strat == "unknown":
            n_unknown += 1
        if not ind:
            industry_miss.append(f"{code} {name}")
        print(f"  → pool {code} {name}[{state}] strategy={strat} industry={ind or '(空)'}")
        if a.apply:
            with storage._conn() as con:
                con.execute("UPDATE pool SET strategy=?, industry=?, updated_at=? WHERE id=?",
                            (strat, ind, dt.datetime.now().isoformat(timespec='seconds'), rid))
        n_pool += 1

    for tbl, key in (("positions", "buy_date"), ("trades", "buy_date")):
        with storage._conn() as con:
            rows = [dict(r) for r in con.execute(f"SELECT * FROM {tbl}")]
        for r in rows:
            if r.get("strategy"):
                continue
            strat = _strategy_of(r.get(key))
            if strat == "unknown":
                n_unknown += 1
            print(f"  → {tbl} #{r['id']} {r.get('code')} strategy={strat}")
            if a.apply:
                with storage._conn() as con:
                    con.execute(f"UPDATE {tbl} SET strategy=? WHERE id=?", (strat, r["id"]))
            if tbl == "positions":
                n_pos += 1
            else:
                n_trd += 1

    print(f"\n═══ 对账报告 ═══")
    print(f"pool 处理 {n_pool} 行 / positions {n_pos} / trades {n_trd} / unknown {n_unknown}")
    if industry_miss:
        print(f"行业缺失 {len(industry_miss)} 只(留空,买入卡按'行业未知'放行): {', '.join(industry_miss)}")
    print("完成" + ("(已写库)" if a.apply else "(dry-run,未写库)"))


if __name__ == "__main__":
    main()
```

```python
# ── P0 行业查询(集中度闸门用):Tushare stock_basic 全市场一次拉取,日级缓存 ──
_INDUSTRY_CACHE = {"date": None, "map": {}, "fail_until": 0.0}


def _industry_map_fetch():
    """全市场 代码→行业名。Tushare stock_basic 的 industry 字段(申万口径)。"""
    pro = _tushare_pro()
    df = pro.stock_basic(exchange="", list_status="L", fields="symbol,industry")
    return {str(r["symbol"]).zfill(6): (r["industry"] or None)
            for _, r in df.iterrows()}


def get_industry(code):
    """个股行业名。主源 Tushare 目录(日缓存+失败600s冷却),备源东财 profile;
    都拿不到返回 None——调用方必须按"未知"放行并如实标注,不许猜。"""
    code = str(code).zfill(6)
    c = _INDUSTRY_CACHE
    today = today_cn().isoformat()
    if c["date"] != today and time.time() >= c["fail_until"]:
        try:
            c["map"] = _industry_map_fetch()
            c["date"] = today
        except Exception as e:
            logger.warning("行业目录拉取失败(600s后重试):%s", e)
            c["fail_until"] = time.time() + 600
    ind = c["map"].get(code)
    if ind:
        return ind
    try:
        prof = get_stock_profile(code)
        return (prof or {}).get("行业") or None
    except Exception:
        return None
```

```python
def industry_exposure(industry):
    """同行业持仓成本合计(元):state='holding' 且 industry 相同的 position_amt 之和。
    口径=买入成本(pool 不存股数,算不了实时市值);industry 为空返回 0——
    未知行业不计入,调用方必须按"未知"如实标注放行,不许把未知当成 0 风险。"""
    if not industry:
        return 0.0
    with _conn() as con:
        row = con.execute(
            "SELECT COALESCE(SUM(position_amt),0) AS s FROM pool "
            "WHERE state='holding' AND industry=?", (industry,)).fetchone()
        return float(row["s"] or 0)
```

```python
def _update_circuit_breaker(pos_alerts, trade_date):
    """P0 账户熔断(设计§3.1):权益=账户总额+已实现盈亏+持仓浮动盈亏(成本×现价/入场价),
    自高点回撤达 DD_HALT_PCT 即熔断——只停新开仓(记账端点硬挡),持仓管理照常;
    恢复必须设置页人工确认。未设账户总额→返回 None(无基准不误报)。"""
    try:
        acct = storage.load_settings().get("ACCOUNT_SIZE")
        acct = float(acct) if acct else None
    except Exception:
        acct = None
    if not acct:
        return None
    realized = float(storage.trade_stats().get("pnl_total") or 0)
    closes = {p.get("code"): p.get("close") for p in (pos_alerts or []) if p.get("close")}
    float_pnl = 0.0
    for r in storage.pool_list("holding"):
        amt, ep, c = r.get("position_amt"), r.get("entry_price"), closes.get(r.get("code"))
        if amt and ep and c:
            float_pnl += float(amt) * (float(c) / float(ep) - 1.0)
    equity = acct + realized + float_pnl
    try:
        peak = float(storage.kv_get("equity_peak") or 0)
    except (TypeError, ValueError):
        peak = 0.0
    if equity > peak:
        peak = equity
        storage.kv_set("equity_peak", f"{peak:.2f}")
    dd = (equity / peak - 1.0) * 100 if peak > 0 else 0.0
    halt_pct = float(params.current_values().get("DD_HALT_PCT", 8.0))
    already = storage.kv_get("circuit_breaker") == "on"
    if dd <= -halt_pct and not already:
        storage.kv_set("circuit_breaker", "on")
        storage.event_add("000000", trade_date, "circuit_breaker",
                          f"账户自高点回撤 {dd:.1f}%,触发熔断:停止一切新开仓",
                          payload={"equity": round(equity, 2), "peak": round(peak, 2),
                                   "dd_pct": round(dd, 2), "halt_pct": halt_pct})
        try:
            push.send_push("⛔ 账户熔断",
                           f"自高点回撤 {dd:.1f}%(熔断线 -{halt_pct:.0f}%),已停止一切新开仓;"
                           f"持有票照常按卖出线管理。恢复开仓需在 设置→维修间 人工确认。")
        except Exception:
            pass
    return {"equity": round(equity, 2), "peak": round(peak, 2),
            "dd_pct": round(dd, 2), "halted": storage.kv_get("circuit_breaker") == "on"}
```
