```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
简明模式翻译层(v3 任务7):把规则引擎/日报/池子的输出翻译成大白话消息卡。
纪律:本模块是决定性纯函数,不碰网络、不调AI、不改任何状态;
消息卡里的每个数字都原样来自输入数据——AI 只在「问AI」页讲课,不进这里。
"""
import math

# 黑话→人话词典(设计稿定稿,全站唯一来源)
STATE_CN = {"candidate": "新发现", "research": "研究中", "watchlist": "等买点",
            "holding": "持有中", "cooldown": "冷静期", "removed": "已移出"}

RISK_PCT = 2.0        # 单笔风险预算:账户的2%(规则卡仓位纪律)
POS_CAP_PCT = 30.0    # 单票仓位上限30%(规则卡)
DD_EXIT = 0.25        # E3:自持有期最高收盘回撤25%(与 rules_mid CONFIG_MID 口径一致)


def sell_line(pos_alert, pool_row):
    """当前有效卖出线(展示用,判定仍在规则引擎):
    取 有效止损线(E1,含上移)/锁盈线(E3=持有高点×0.75)/生命线(E2=周MA30) 三者最高。
    locked:线是否已抬过成本(breakeven)/高出成本5%以上(profit)。"""
    pa = pos_alert or {}
    row = pool_row or {}
    entry = pa.get("entry_price") or row.get("entry_price")
    cands = []
    eff_stop = pa.get("eff_stop") or pa.get("stop_price") or row.get("stop_price") \
        or row.get("platform_price")
    if eff_stop:
        cands.append((float(eff_stop), "止损线"))
    if pa.get("held_peak"):
        cands.append((round(float(pa["held_peak"]) * (1 - DD_EXIT), 2), "锁盈线"))
    if pa.get("week_ma30"):
        cands.append((round(float(pa["week_ma30"]), 2), "生命线"))
    if not cands:
        return {"price": None, "basis": None, "locked": "none"}
    price, basis = max(cands, key=lambda x: x[0])
    locked = "none"
    if entry:
        entry = float(entry)
        if price >= entry * 1.05:
            locked = "profit"
        elif price >= entry:
            locked = "breakeven"
    if basis == "止损线" and entry and eff_stop and float(eff_stop) >= entry:
        basis = "保本线"
    return {"price": round(price, 2), "basis": basis, "locked": locked}


def suggest_position(account_size, close, stop):
    """建议买入金额:单笔最多亏账户 RISK_PCT%,金额不超过账户 POS_CAP_PCT%。
    股数取整到100股;算不出(缺数据/风险距离≤0)给人话 note。"""
    if not account_size:
        return {"amount": None, "shares": None,
                "note": "在设置里填一次账户总额,以后帮您算买多少"}
    try:
        close, stop = float(close), float(stop)
        account = float(account_size)
    except (TypeError, ValueError):
        return {"amount": None, "shares": None, "note": "价格数据缺失,金额请自行把握"}
    risk_per_share = close - stop
    if risk_per_share <= 0:
        return {"amount": None, "shares": None,
                "note": "止损线不在现价下方,该信号金额请自行把握"}
    shares = (account * RISK_PCT / 100) / risk_per_share          # 风险倒算
    shares = min(shares, account * POS_CAP_PCT / 100 / close)     # 30%仓位帽
    shares = int(shares // 100 * 100)                             # 100股取整
    if shares < 100:
        return {"amount": None, "shares": None,
                "note": "按2%风险算不足100股,这单要么放弃要么金额自行把握"}
    amount = int(round(shares * close))
    return {"amount": amount, "shares": shares,
            "note": f"按您的规矩:单笔最多亏账户的{RISK_PCT:.0f}%,倒算出这个数"}
```

### Task 3: brief.build_brief — 消息卡组装

**Files:**
- Modify: `brief.py`
- Test: `test_web.py`（T7B1~T7B7）

**Interfaces:**
- Consumes: Task 1 的 `sell_line/suggest_position/STATE_CN`。
- Produces: `build_brief(report, pool_rows, snap, account_size) -> dict`：
  `{"date": str, "n_actions": int, "cards": [CARD...], "holdings": [...], "waiting": [...]}`
  CARD = `{"kind": "sell|trim|buy|new|review", "code", "name", "headline": str, "facts": [[label,value]...], "plain": str, "howto": str, "actions": [{"id","label"}], "learn": {"title": str, "paras": [str...]}|None}`
  排序:sell → trim → buy → review → new。
  HOLDING行 = `{"code","name","pct_chg","close","sell_line":{...},"light":"ok|warn|bad","note":str}`
  映射规则（数字全部原样搬运）:
  - next_action.type EXIT → sell 卡（learn=止损课,固定文案模板）
  - TRIM → trim 卡（含"卖出一半(约N股)"——N=persisted position_amt 折算,缺则不显示股数;learn=落袋课）
  - ENTRY_READY/ENTER → buy 卡（facts 含买点区间[platform, close]、suggest_position 金额/股数、止损线、"最多亏账户2%"、"赚了后卖出线自动上移";learn 三段:技术面模板(用 tech hint/数字)+基本面(thesis bull_case+最新 earnings 事件 title)+风险(bear_report 三路径)）
  - REVIEW → review 卡（事件标题+ai_note 结论人话:验证→"支持当初买它的理由,不用动"/削弱→"动摇了买它的理由,建议您看一眼"）
  - pool state=candidate 且无 thesis → new 卡（actions: research_yes/research_no）
  - HOLD/WATCH → 不出卡,进 holdings/waiting 列表。
  - holdings.light: 有 EXIT 卡→bad;有 TRIM 卡→warn;否则 ok。waiting 行的 note 用 next_action.condition 原文。
